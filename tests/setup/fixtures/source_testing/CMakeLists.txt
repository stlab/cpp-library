cmake_minimum_required(VERSION 3.24)
include("${TOOLKIT_SOURCE}/cpp-library.cmake")
project(sample LANGUAGES CXX)
include(CTest)
set(CPP_LIBRARY_VERSION 1.0.0)
set(FIXTURE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)

function(CPMAddPackage specification)
    if(NOT specification MATCHES "^gh:doctest/doctest@([0-9.]+)$")
        message(FATAL_ERROR "Unexpected test dependency: ${specification}")
    endif()
    set(version "${CMAKE_MATCH_1}")
    include(FetchContent)
    FetchContent_Declare(doctest
        URL "https://github.com/doctest/doctest/archive/refs/tags/v${version}.tar.gz")
    FetchContent_MakeAvailable(doctest)
endfunction()

if(NOT CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 20)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(source_args SOURCES component.cpp)
if(HEADER_ONLY)
    set(source_args)
endif()
set(test_args TESTS existing.cpp rejected_fail.cpp EXAMPLES example.cpp)
if(SOURCE_ONLY)
    set(test_args)
endif()
cpp_library_setup(DESCRIPTION "Public and in-source testing" NAMESPACE fixture
    HEADERS api.hpp REQUIRES_CPP_VERSION "${CMAKE_CXX_STANDARD}" ${source_args}
    ${test_args})

foreach(test_target existing rejected_fail)
    if(TARGET "${test_target}")
        get_target_property(links "${test_target}" LINK_LIBRARIES)
        if(NOT "fixture::sample" IN_LIST links)
            message(FATAL_ERROR "Public test ${test_target} must link the production library: ${links}")
        endif()
    endif()
endforeach()
if(TARGET existing AND BUILD_SHARED_LIBS AND NOT HEADER_ONLY)
    target_compile_definitions(existing PRIVATE EXPECT_SHARED)
endif()

add_library(requirements INTERFACE)
target_compile_definitions(requirements INTERFACE PUBLIC_VALUE=19)
if(HEADER_ONLY)
    target_compile_definitions(sample INTERFACE FIXTURE_HEADER_ONLY)
    target_link_libraries(sample INTERFACE requirements)
    return()
endif()
set_target_properties(sample PROPERTIES CXX_VISIBILITY_PRESET hidden)
target_sources(sample PRIVATE src/registration.cpp)
target_compile_definitions(sample PRIVATE PRIVATE_VALUE=23
    "REQUIRED_STANDARD=${CMAKE_CXX_STANDARD}"
    "EXPECTED_INTERNAL=$<IF:$<BOOL:${FAIL_EMBEDDED}>,41,42>")
target_link_libraries(sample PUBLIC requirements)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(sample INTERFACE FIXTURE_USING_DLL)
endif()
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated/config.hpp")
add_custom_command(OUTPUT "${generated}"
    COMMAND "${CMAKE_COMMAND}" "-DOUTPUT=${generated}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/generate.cmake")
if(FILE_SET_GENERATION)
    if(DEFAULT_HEADER_SET)
        target_sources(sample PUBLIC FILE_SET HEADERS
            BASE_DIRS "${CMAKE_CURRENT_BINARY_DIR}/generated" FILES "${generated}")
    else()
        target_sources(sample PRIVATE FILE_SET generated_headers TYPE HEADERS
            BASE_DIRS "${CMAKE_CURRENT_BINARY_DIR}/generated" FILES "${generated}")
    endif()
else()
    add_custom_target(generate_config DEPENDS "${generated}")
    add_dependencies(sample generate_config)
endif()
target_include_directories(sample PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(NOT BUILD_TESTING OR NOT PROJECT_IS_TOP_LEVEL)
    foreach(target existing rejected_fail sample_in_source_tests doctest::doctest)
        if(TARGET "${target}")
            message(FATAL_ERROR "Unexpected development dependency/target: ${target}")
        endif()
    endforeach()
endif()
